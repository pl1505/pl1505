    public partial class RegisterPage : Page
    {
        public RegisterPage()
        {
            InitializeComponent();
            LoadRoles();
        }
        private void LoadRoles()
        {
            try
            {
                List<Role> roles = AppData.db.Role.ToList();
                CmbRole.ItemsSource = roles;
                CmbRole.DisplayMemberPath = "Title";   
                CmbRole.SelectedValuePath = "ID";      
                if (roles.Any()) 
                {
                    CmbRole.SelectedIndex = 1; 
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка при загрузке ролей: {ex.Message}");
            }
        }
        private void BtnRegistr_Click(object sender, RoutedEventArgs e)
        {
            var existingUser = AppData.db.User
                         .FirstOrDefault(u => u.Login == TxbLogin.Text.Trim());

            if (existingUser != null)
            {
                MessageBox.Show("Пользователь с таким логином уже существует!");
                return;
            }

           
            if (CmbRole.SelectedValue is int roleId)
            {
                var newUser = new User
                {
                    Login = TxbLogin.Text.Trim(),
                    Password = TxbPassword.Text,
                    Name = Name_info.Text,
                    FirstName = Family_info.Text,
                    MiddleName = MiddleName_info.Text,
                    Inn = Inn_info.Text,
                    Description = Description_Info.Text,
                    DatePicker = DatePicker_info.Text,
                    RoleID = roleId 
                };
                AppData.db.User.Add(newUser);
                AppData.db.SaveChanges();
                MessageBox.Show("Регистрация успешна! Теперь вы можете войти.");
            }
            else
            {
                MessageBox.Show("Выберите роль для пользователя.");
            }

        }

        private void BtnLogin_Click(object sender, RoutedEventArgs e)
        {
            NavigationService.Navigate(new SignlnPage());
            
        }
    }
